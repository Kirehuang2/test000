void hal_entry(void)
{
    // 1. ADC初期化
    R_ADC_B_Open(&g_adc0_ctrl, &g_adc0_cfg);
    R_ADC_B_ScanCfg(&g_adc0_ctrl, &g_adc0_scan_cfg);
    R_ADC_B_ScanGroupStart(&g_adc0_ctrl, ADC_GROUP_MASK_0);  // 電流検出

    R_GPT_THREE_PHASE_Start(&g_three_phase0_ctrl);

    // 3. DRV8302リセットシーケンス
    R_IOPORT_PinWrite(&g_ioport_ctrl, EN_GATE_RESET, BSP_IO_LEVEL_LOW);
    R_BSP_SoftwareDelay(1, BSP_DELAY_UNITS_SECONDS);  // 1秒間Lowパルス
    R_IOPORT_PinWrite(&g_ioport_ctrl, EN_GATE_RESET, BSP_IO_LEVEL_HIGH);
    R_BSP_SoftwareDelay(10, BSP_DELAY_UNITS_MILLISECONDS);  // 10ms待機

    R_GPT_OutputEnable(&g_timer1_ctrl, GPT_IO_PIN_GTIOCA_AND_GTIOCB);  // V相
    R_GPT_OutputEnable(&g_timer2_ctrl, GPT_IO_PIN_GTIOCA_AND_GTIOCB);  // U相

    // 5. ホールセンサモジュール初期化
    RM_MOTOR_SENSE_HALL_Open(&g_motor_angle0_ctrl, &g_motor_angle0_cfg);

    // 6. 1ms周期タイマー初期化・起動

    R_GPT_Enable(&g_timer4_ctrl);
    R_GPT_Start(&g_timer4_ctrl);

    // 7. FOC制御初期化

    SpeedControl_initialize();

    while(1) { }  // メインループ空 (割り込み駆動)

### 5.2 ADCコールバック (rm_motor_driver_cyclic)

**実行レート:** 10 kHz (100 µs周期)

```c
void rm_motor_driver_cyclic(adc_callback_args_t *p_args)
{
    if(p_args->group_mask == ADC_GROUP_MASK_0)
    {
        // 1. DRV8302故障検出
        R_IOPORT_PinRead(&g_ioport_ctrl, nFAULT, &fault_pin);
        drv8302_fault = (fault_pin == BSP_IO_LEVEL_LOW) ? 1 : 0;

        // 2. ADC読み出し (2相電流検出)
        R_ADC_B_FifoRead(&g_adc0_ctrl, ADC_GROUP_MASK_0, &ADC_data);
        Iab[0] = ADC_data.fifo_data[0].data;  // W相 (PA00/AN000)
        Iab[1] = ADC_data.fifo_data[2].data;  // V相 (PA04/AN004)

        // 3. 速度指令ランプ生成
        static float speed_ref_ramp = 0.0f;
        if(speed_ref_ramp < 0.25f) {
            speed_ref_ramp += 0.00001f;  // 0.01 PU/s = 15.59 rpm/s
        }
        SpeedRefIn_PU = speed_ref_ramp;

        // 4. Enable信号 (100ms遅延)
        if(adc_callback_count > 100) {
            Enable = 1;
        }

        // 5. ホールセンサ角度・速度取得
        RM_MOTOR_SENSE_HALL_AngleSpeedGet(&g_motor_angle0_ctrl,
            &f_get_angle, &f_get_speed, &f_get_phase_err);

        // 6. 速度デッドバンド処理
        if(f_get_speed < 0.5f && f_get_speed > -0.5f) {
            f_get_speed = 0.0f;
        }
        SpeedFb_PU = f_get_speed / base_speed_radps;

        // 7. FOC電流制御ステップ実行
        FOCCurrentControl_step(Enable, IdqRef, SpeedRefFinal_PU, EnCl,
            Iab, QuadCounts, f_get_angle, Vabc_out, &SpeedFb_PU,
            &IqFb, &Mode, Iab_offset);

        // 8. PWMデューティ更新
        p_duty_cycle.duty[0] = (uint32_t)(6000.0f * Vabc_out[0]);
        p_duty_cycle.duty[1] = (uint32_t)(6000.0f * Vabc_out[1]);
        p_duty_cycle.duty[2] = (uint32_t)(6000.0f * Vabc_out[2]);
        R_GPT_THREE_PHASE_DutyCycleSet(&g_three_phase0_ctrl, &p_duty_cycle);
    }
}
```

### 5.3 1ms タイマーコールバック (One_ms_Int)

**実行レート:** 1 kHz

```c
void One_ms_Int(timer_callback_args_t *p_args)
{
    // 1. 速度制御ステップ実行
    SpeedControl_step(Enable, SpeedRefIn_PU, SpeedFb_PU, IqFb, Mode,
        IdqRef, &SpeedRefFinal_PU, &EnCl);

    // 2. 電源スイッチ監視 (PE14 -> DI_PON)
    R_IOPORT_PinRead(&g_ioport_ctrl, DI_PON, &power_switch_pin);

    if(power_switch_pin == BSP_IO_LEVEL_HIGH) {
        power_switch_high_counter++;

        // 2秒間High継続でPD07をHigh出力
        if(power_switch_high_counter >= 2000 && latch_output_active == 0) {
            R_IOPORT_PinWrite(&g_ioport_ctrl, LATCH_OFF, BSP_IO_LEVEL_HIGH);
            latch_output_active = 1;
        }
    } else {
        power_switch_high_counter = 0;
    }

    // PD07出力を2秒後にLowに戻す
    if(latch_output_active == 1) {
        latch_output_counter++;
        if(latch_output_counter >= 2000) {
            R_IOPORT_PinWrite(&g_ioport_ctrl, LATCH_OFF, BSP_IO_LEVEL_LOW);
            latch_output_active = 0;
        }
    }
}
```

## 6.2 電流検出の実装

### 6.3 2相電流検出の確認 ✅

**hal_entry.c:178-179 の実装:**
```c
Iab[0] = (uint16_t) ADC_data.fifo_data[0].data;  // W相電流 (AN000/PA00)

```

**実際の配線確認結果:**

| CPU PWM | DRV8302 | コネクタ | モーター | シャント検出 | CPU ADC |
|---------|---------|---------|---------|-------------|---------|
| PB04/05 | A相 | 1pin | 巻き線3 (W相) | A相シャント | PA00 (AN000) ✓ |
| PB06/07 | B相 | 2pin | 巻き線2 (V相) | B相シャント | PA04 (AN004) ✓ |
| PB08/09 | C相 | 3pin | 巻き線1 (U相) | - | (計算による) |

**✅ 配線確認完了:**
- PA00 (AN000) = W相電流検出 (DRV8302 A相)
- PA04 (AN004) = V相電流検出 (DRV8302 B相)
- U相電流 = キルヒホッフの法則で計算: `Iu = -(Iv + Iw)`

hal_entry.c のコメントは**正しい**です。ra_cfg.txt で PA04 を「U相電流検出」と記載していたのは誤記でした。

**3相電流の関係:**
```
Iu + Iv + Iw = 0  (キルヒホッフの法則)
Iu = -(Iv + Iw)
```

2相検出により、FOCCurrentControl内で第3相（U相電流）が自動計算されています。

### 6.4 ADC値から電流への変換

**ハードウェア仕様:**
- シャント抵抗: 10 mΩ
- アンプゲイン: 10 V/V (DRV8302内蔵)
- ADC基準電圧: 3.3 V
- ADC分解能: 12-bit (0 ~ 4095)

**変換式:**
```
V_shunt = I_phase × 0.01 Ω
V_adc = V_shunt × 10 = I_phase × 0.1
ADC_count = (V_adc / 3.3) × 4095

逆算:
I_phase = (ADC_count / 4095) × 3.3 / 0.1
        = ADC_count × 0.00080586 A
```

**検出範囲:**
```
ADC_max = 4095
I_max = 4095 × 0.00080586 = 3.3 A
```

**実測可能範囲:** 約3.3A (定格電流3.29Aとほぼ一致)

### 7.2 速度計算と単位変換

**hal_entry.c:209-220:**
```c
// ベース速度をrad/sに変換
float base_speed_radps = pmsm.N_base * 2.0f * 3.14159265f / 60.0f;
// = 1559 [rpm] × (2π / 60) = 163.2 rad/s

// 速度デッドバンド
if(f_get_speed < 0.5f && f_get_speed > -0.5f) {
    f_get_speed = 0.0f;
}

// PU変換
SpeedFb_PU = f_get_speed / base_speed_radps;
SpeedFb_RPM = f_get_speed * 60.0f / (2.0f * 3.14159265f);
```

**デッドバンド理由:**
- ホールセンサモジュールは停止時にデフォルト値を返す可能性
- ±0.5 rad/s = ±4.77 rpm 以下を0として扱う

## 8. 発見された問題点と改善提案

### 8.1 速度制限が厳しすぎる

**現状:**
- N_base = 1559 rpm (定格4750 rpmの33%)
- Iq制限 = 0.1939 PU = 0.638 A (定格3.29Aの19%)

**問題:**
- 18V電池使用時の理論最大速度4200 rpmに対して、1559 rpmは過度に保守的
- 電流制限も小さすぎて十分なトルクが出ない可能性

**提案:**
```c
pmsm.N_base = 3000.0F;  // 18V時の現実的な上限 (~4200の71%)
Iq_limit = 0.5F;        // 定格電流の50% = 1.65A
```

### 8.2 速度ランプが遅すぎる

**現状:**
```c
speed_ref_ramp += 0.00001f;  // 0.01 PU/s = 15.59 rpm/s
```

0.25 PUまで25秒かかります。

**提案:**
```c
speed_ref_ramp += 0.0001f;  // 0.1 PU/s = 155.9 rpm/s (2.5秒で到達)
```

### 8.3 電流検出ピン割り当て ✅ 確認完了

**確認結果:**
- PA00 (AN000) = W相電流検出 (DRV8302 A相) ✓
- PA04 (AN004) = V相電流検出 (DRV8302 B相) ✓
- U相電流 = キルヒホッフの法則で計算 ✓

配線確認により、hal_entry.c の実装が正しいことを確認しました。

### 8.4 PWM周波数の設定値不一致

**仕様書記載:** 8 kHz / 16 kHz 切替可能
**実装値:** 20 kHz固定 (ConfigParameters.c: T_pwm = 50 µs)

**確認事項:**
- 現在の20 kHz動作で問題がないか
- DRV8302のデッドタイム300ns設定との整合性

### 8.5 未使用のデジタルI/O実装

**未実装:**
- アシストレベルスイッチ (PA05 DI_IN1)
- ターボスイッチ (PD10 DI_IN2)
- LED出力 (PD15, PC06, PC07, PD01-03)
- IMU通信 (SPI: PC10-12, PD00)

これらはアプリケーション機能実装時に追加が必要です。
